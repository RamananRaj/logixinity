Database Setup

Run this SQL once in your Supabase dashboard (SQL Editor) to create the blog posts table.

In Supabase → SQL Editor, paste and run the SQL below. Then go to Authentication → Policies and enable RLS with the policies shown.
-- 1. Create the blog_posts table
CREATE TABLE IF NOT EXISTS blog_posts (
  id          uuid        DEFAULT gen_random_uuid() PRIMARY KEY,
  title       text        NOT NULL,
  content     text        NOT NULL DEFAULT '',
  cover_image text,
  author      text,
  publish_date date,
  tags        text,
  published   boolean     DEFAULT false,
  created_at  timestamptz DEFAULT now(),
  updated_at  timestamptz DEFAULT now()
);

-- 2. Enable Row Level Security
ALTER TABLE blog_posts ENABLE ROW LEVEL SECURITY;

-- 3. Public can read published posts (for blog.html)
CREATE POLICY "Public read published"
  ON blog_posts FOR SELECT
  USING (published = true);

-- 4. Anon can do everything (admin page uses anon key)
--    In production, restrict this to authenticated users
CREATE POLICY "Anon full access"
  ON blog_posts FOR ALL
  USING (true)
  WITH CHECK (true);
